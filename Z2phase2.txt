<?php
session_start();
include 'db_connection.php';

if (!isset($_SESSION['user_id'], $_SESSION['flight_id'], $_SESSION['class_type'], $_SESSION['passengers'], $_SESSION['total_amount'])) {
    echo "Session expired. Please restart your booking.";
    exit;
}

$user_id = $_SESSION['user_id'];
$flight_id = $_SESSION['flight_id'];
$return_flight_id = $_SESSION['return_flight_id'] ?? null;
$class_type = $_SESSION['class_type'];
$total_amount = $_SESSION['total_amount'];
$passengers = $_SESSION['passengers'];

if (isset($_POST['pay_now'])) {
    try {
        // Start transaction
        $pdo->beginTransaction();

        // 1. Insert into booking
        $stmt = $pdo->prepare("INSERT INTO bookings2 (user_id, flight_id, return_flight_id, class_type, total_amount, payment_status) VALUES (:user_id, :flight_id, :return_flight_id, :class_type, :total_amount, 'Paid')");
        $stmt->bindParam(':user_id', $user_id, PDO::PARAM_INT);
        $stmt->bindParam(':flight_id', $flight_id, PDO::PARAM_INT);
        $stmt->bindParam(':return_flight_id', $return_flight_id, PDO::PARAM_INT);
        $stmt->bindParam(':class_type', $class_type, PDO::PARAM_STR);
        $stmt->bindParam(':total_amount', $total_amount, PDO::PARAM_STR); // Assuming total_amount is stored as string or decimal
        $stmt->execute();
        $booking_id = $pdo->lastInsertId();
        $stmt->closeCursor();

        // 2. Insert passengers
        $stmt_passenger = $pdo->prepare("INSERT INTO passenger (booking_id, name, gender, age, seat_no) VALUES (:booking_id, :name, :gender, :age, '')");
        foreach ($passengers as $p) {
            $stmt_passenger->bindParam(':booking_id', $booking_id, PDO::PARAM_INT);
            $stmt_passenger->bindParam(':name', $p['name'], PDO::PARAM_STR);
            $stmt_passenger->bindParam(':gender', $p['gender'], PDO::PARAM_STR);
            $stmt_passenger->bindParam(':age', $p['age'], PDO::PARAM_INT);
            $stmt_passenger->execute();
        }
        $stmt_passenger->closeCursor();

        // 3. Dummy payment insert
        $payment_mode = "Razorpay";
        $transaction_id = uniqid('txn_');
        $stmt_payment = $pdo->prepare("INSERT INTO payments (booking_id, payment_mode, transaction_id, amount) VALUES (:booking_id, :payment_mode, :transaction_id, :amount)");
        $stmt_payment->bindParam(':booking_id', $booking_id, PDO::PARAM_INT);
        $stmt_payment->bindParam(':payment_mode', $payment_mode, PDO::PARAM_STR);
        $stmt_payment->bindParam(':transaction_id', $transaction_id, PDO::PARAM_STR);
        $stmt_payment->bindParam(':amount', $total_amount, PDO::PARAM_STR); // Assuming amount is stored as string or decimal
        $stmt_payment->execute();
        $stmt_payment->closeCursor();

        // 4. Update seat count for flight(s)
        $seats_to_reduce = count($passengers);
        $column_to_update = $class_type . "_seats";
        $update_flight_sql = "UPDATE flights2 SET $column_to_update = $column_to_update - :seats_to_reduce WHERE id = :flight_id";
        $stmt_update_flight = $pdo->prepare($update_flight_sql);
        $stmt_update_flight->bindParam(':seats_to_reduce', $seats_to_reduce, PDO::PARAM_INT);
        $stmt_update_flight->bindParam(':flight_id', $flight_id, PDO::PARAM_INT);
        $stmt_update_flight->execute();
        $stmt_update_flight->closeCursor();

        if ($return_flight_id) {
            $stmt_update_return_flight = $pdo->prepare($update_flight_sql);
            $stmt_update_return_flight->bindParam(':seats_to_reduce', $seats_to_reduce, PDO::PARAM_INT);
            $stmt_update_return_flight->bindParam(':flight_id', $return_flight_id, PDO::PARAM_INT);
            $stmt_update_return_flight->execute();
            $stmt_update_return_flight->closeCursor();
        }

        // Commit transaction
        $pdo->commit();

        // 5. Clear session except user info, redirect to success
        $_SESSION['booking_id'] = $booking_id;
        unset($_SESSION['flight_id'], $_SESSION['return_flight_id'], $_SESSION['class_type'], $_SESSION['passengers'], $_SESSION['total_amount']);

        header("Location: success_booking.php");
        exit;

    } catch (PDOException $e) {
        // Rollback transaction on error
        if ($pdo->inTransaction()) {
            $pdo->rollBack();
        }
        echo "Error processing payment: " . $e->getMessage();
        // Optionally log the error for debugging
    }
}

    <p>Payment Mode: Razorpay (dummy)</p>

                            WHERE b.id = :booking_id";

try {
    // Fetch user profile
    $user_query = "SELECT * FROM users WHERE id = :user_id";
    $stmt_user = $pdo->prepare($user_query);
    $stmt_user->bindParam(':user_id', $user_id, PDO::PARAM_INT);
    $stmt_user->execute();
    $user = $stmt_user->fetch(PDO::FETCH_ASSOC);
    $stmt_user->closeCursor();

    // Current bookings (future flights with payment done)
    $current_query = "
        SELECT b.*, f.from_location AS source, f.to_location AS destination, f.departure_time
        FROM bookings2 b
        JOIN flights2 f ON b.flight_id = f.id
        WHERE b.user_id = :user_id
          AND b.payment_status = 'Paid'
          AND f.departure_time > NOW()
        ORDER BY f.departure_time ASC
    ";
    $stmt_current = $pdo->prepare($current_query);
    $stmt_current->bindParam(':user_id', $user_id, PDO::PARAM_INT);
    $stmt_current->execute();
    $current_bookings = $stmt_current->fetchAll(PDO::FETCH_ASSOC);
    $stmt_current->closeCursor();

    // Travel history (past flights)
    $history_query = "
        SELECT b.*, f.from_location AS source, f.to_location AS destination, f.departure_time
        FROM bookings2 b
        JOIN flights2 f ON b.flight_id = f.id
        WHERE b.user_id = :user_id
          AND b.payment_status = 'Paid'
          AND f.departure_time <= NOW()
        ORDER BY f.departure_time DESC
    ";
    $stmt_history = $pdo->prepare($history_query);
    $stmt_history->bindParam(':user_id', $user_id, PDO::PARAM_INT);
    $stmt_history->execute();
    $travel_history = $stmt_history->fetchAll(PDO::FETCH_ASSOC);

} catch (PDOException $e) {
