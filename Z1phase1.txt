<?php
session_start();
include 'includes/header.php';
require_once 'includes/db.php'; // Include your PDO DB connection

// Redirect if already logged in
if (isset($_SESSION['user_id'])) {
    header("Location: user_dashboard.php");
    exit;
}

$error = '';
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $name     = trim($_POST['name']);
    $email    = trim($_POST['email']);
    $phone    = trim($_POST['phone']);
    $password = $_POST['password'];
    $confirm  = $_POST['confirm'];

    if ($password !== $confirm) {
        $error = "Passwords do not match!";
    } elseif (!empty($name) && !empty($email) && !empty($phone) && !empty($password)) {
        // Check if email already exists
        $stmt = $pdo->prepare("SELECT id FROM users WHERE email = ?");
        $stmt->execute([$email]);
        if ($stmt->rowCount() > 0) {
            $error = "Email already registered!";
        } else {
            // Hash and store the password
            $hashedPwd = password_hash($password, PASSWORD_BCRYPT);
            $insert = $pdo->prepare("INSERT INTO users (name, email, password, phone) VALUES (?, ?, ?, ?)");
            if ($insert->execute([$name, $email, $hashedPwd, $phone])) {
                // Auto-login
                $user_id = $pdo->lastInsertId();
                $_SESSION['user_id']   = $user_id;
                $_SESSION['user_name'] = $name;

                // Redirect appropriately
                if (isset($_SESSION['booking_data']) && isset($_SESSION['passengers'])) {
                    header("Location: passenger_details.php");
                } else {
                    header("Location: user_dashboard.php");
                }
                exit;
            } else {
                $error = "Something went wrong. Please try again.";
            }
        }
    } else {
        $error = "All fields are required!";
    }
}

<?php
session_start();
include 'includes/header.php';
require_once 'includes/db.php'; // Include your PDO DB connection

// Check if user is logged in, if not redirect to login page
if (!isset($_SESSION['user_id'])) {
    header("Location: login.php");
    exit();
}

// Get form data from flight_search.php
$from = $_GET['from'];
$to = $_GET['to'];
$departure_date = $_GET['departure_date'];
$trip_type = $_GET['trip_type'];
$class_type = $_GET['class_type'];
$min_price = $_GET['min_price'];
$max_price = $_GET['max_price'];

// Create a dynamic SQL query to search flights based on user inputs
$sql = "SELECT * FROM flights2 WHERE from_location = :from AND to_location = :to AND DATE(departure) = :departure_date";
$params = [
    ':from' => $from,
    ':to' => $to,
    ':departure_date' => $departure_date,
];

// Add price filtering
if ($min_price != '') {
    $sql .= " AND (economy_price >= :min_price OR business_price >= :min_price OR first_class_price >= :min_price)";
    $params[':min_price'] = $min_price;
}
if ($max_price != '') {
    $sql .= " AND (economy_price <= :max_price OR business_price <= :max_price OR first_class_price <= :max_price)";
    $params[':max_price'] = $max_price;
}

try {
    // Prepare and execute the SQL query
    $stmt = $pdo->prepare($sql);
    $stmt->execute($params);
    $result = $stmt->fetchAll(PDO::FETCH_ASSOC);

    // Check if any flights are available
    if ($result) {
        // Display flights
        echo "<h2>Available Flights</h2><table>";
        echo "<tr><th>Airline</th><th>Departure</th><th>Arrival</th><th>Economy Price</th><th>Business Price</th><th>First Class Price</th><th>Book Now</th></tr>";

        foreach ($result as $row) {
            echo "<tr>
                    <td>{$row['airline_name']}</td>
                    <td>{$row['departure']}</td>
                    <td>{$row['arrival']}</td>
                    <td>{$row['economy_price']}</td>
                    <td>{$row['business_price']}</td>
                    <td>{$row['first_class_price']}</td>
                    <td><a href='?book={$row['id']}&class=$class_type' class='book-btn'>Book Now</a></td>
                  </tr>";
        }
        echo "</table>";
    } else {
        echo "<p>No flights found for your search. Please try again with different criteria.</p>";
    }

    // Handle "Book Now" button click
    if (isset($_GET['book'])) {
        $flight_id = $_GET['book'];
        $selected_class = $_GET['class'];

        // Store selected flight, class, and price in session
        $stmt_flight = $pdo->prepare("SELECT * FROM flights2 WHERE id = :flight_id");
        $stmt_flight->bindParam(':flight_id', $flight_id, PDO::PARAM_INT);
        $stmt_flight->execute();
        $flight_data = $stmt_flight->fetch(PDO::FETCH_ASSOC);

        if ($flight_data) {
            $class_price = 0;
            switch ($selected_class) {
                case 'economy':
                    $class_price = $flight_data['economy_price'];
                    break;
                case 'business':
                    $class_price = $flight_data['business_price'];
                    break;
                case 'first':
                    $class_price = $flight_data['first_class_price'];
                    break;
            }

            // Store booking data in session
            $_SESSION['booking_data'] = [
                'flight_id' => $flight_id,
                'class' => $selected_class,
                'price' => $class_price,
                'departure' => $flight_data['departure'],
                'airline' => $flight_data['airline_name']
            ];

            // Redirect to passenger details page
            header("Location: passenger_details.php");
            exit();
        } else {
            echo "<p>Error: Flight not found.</p>";
        }
    }

} catch (PDOException $e) {
    echo "Error: " . $e->getMessage();
}
